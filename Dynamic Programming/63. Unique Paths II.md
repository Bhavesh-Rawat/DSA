
# 63. Unique Paths II

You are given an m x n integer array grid. There is a robot initially located at the top-left corner (i.e., grid[0][0]). The robot tries to move to the bottom-right corner (i.e., grid[m - 1][n - 1]). The robot can only move either down or right at any point in time.

An obstacle and space are marked as 1 or 0 respectively in grid. A path that the robot takes cannot include any square that is an obstacle.

Return the number of possible unique paths that the robot can take to reach the bottom-right corner.

The testcases are generated so that the answer will be less than or equal to 2 * 109.

##Java
```java

class Solution {
    public int uniquePathsWithObstacles(int[][] obstacleGrid) {
         return totalWays (0,0,obstacleGrid.length,obstacleGrid[0].length, new HashMap <String,Integer>(),obstacleGrid);
    }
    public int totalWays (int currentRow, int currentColm, int m, int n, HashMap <String,Integer> memo,int [][] obstacleGrid) {
        // m-1 and n-1 are the target
        if (currentRow >= m  || currentColm >= n || obstacleGrid [currentRow] [currentColm] == 1)
        return 0;

        if (currentRow == m -1 && currentColm == n -1)
        return 1;

         String currentKey = Integer.toString(currentRow) + "_" + Integer.toString(currentColm);
          
        if(memo.containsKey(currentKey))
            return memo.get(currentKey);

        int rightway = totalWays(currentRow, currentColm + 1,m,n, memo,obstacleGrid);

        int downway = totalWays(currentRow + 1,currentColm,m,n, memo,obstacleGrid);

        memo.put(currentKey,(rightway + downway));
        
        return memo.get(currentKey);
    }
}
