
# 931. Minimum Falling Path Sum

Given an n x n array of integers matrix, return the minimum sum of any falling path through matrix.

A falling path starts at any element in the first row and chooses the element in the next row that is either directly below or diagonally left/right. Specifically, the next element from position (row, col) will be (row + 1, col - 1), (row + 1, col), or (row + 1, col + 1).

//Dynamic Prgramming Solution
class Solution {
    public int minFallingPathSum(int[][] matrix) {
        int row = matrix.length;
        int col = matrix[0].length;

        int ans = 1000000;

        for (int currentcol = 0; currentcol < col ; currentcol++)
{
    int tempAns = minPathSum (matrix,0,currentcol,row,col,new HashMap<String,Integer>());
    ans = Math.min(ans,tempAns);
}
    return ans;

               }

    private int minPathSum (int[][] matrix,int currentrow, int currentcol,int row,int col, HashMap<String,Integer> memo) {

        if (currentcol < 0 || currentcol >= col)
        return 1000001;

        if (currentrow == row - 1)
        return matrix[currentrow][currentcol];

        String currentKey = currentcol + "-" + currentrow;

     if (memo.containsKey(currentKey))
     return memo.get(currentKey);

        int left = matrix[currentrow][currentcol] + minPathSum(matrix,currentrow+1,currentcol-1,row,col,memo);
        int right = matrix[currentrow][currentcol] + minPathSum(matrix,currentrow+1,currentcol+1,row,col,memo);
        int down = matrix[currentrow][currentcol] + minPathSum(matrix,currentrow+1,currentcol,row,col,memo);

       memo.put (currentKey,Math.min(left,Math.min(right,down))); 
       return memo.get(currentKey);
    }
}
